46:         uint256 v = a * b;

48:             return v / c;
49:         }

66:         // ceil(transformLiquidityX * sqrtPrice_96 / TwoPower.Pow96) <=
67:         // ceil(floor(amountX * sqrtPrice_96 / TwoPower.Pow96) * sqrtPrice_96 / TwoPower.Pow96) <=
68:         // ceil(amountX * sqrtPrice_96 / TwoPower.Pow96 * sqrtPrice_96 / TwoPower.Pow96) =
69:         // ceil(amountX) = amountX <= uint128.max
