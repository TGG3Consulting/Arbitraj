451:     ) external override returns (int256[] memory) {
452:         // In order to accurately 'simulate' swaps, this function actually does perform the swaps, including calling the
453:         // Pool hooks and updating balances in storage. However, once it computes the final Vault Deltas, it
454:         // reverts unconditionally, returning this array as the revert data.
455:         //
456:         // By wrapping this reverting call, we can decode the deltas 'returned' and return them as a normal Solidity
457:         // function would. The only caveat is the function becomes non-view, but off-chain clients can still call it
458:         // via eth_call to get the expected result.
459:         //
460:         // This technique was inspired by the work from the Gnosis team in the Gnosis Safe contract:
461:         // https://github.com/gnosis/safe-contracts/blob/v1.2.0/contracts/GnosisSafe.sol#L265
462:         //
463:         // Most of this function is implemented using inline assembly, as the actual work it needs to do is not
464:         // significant, and Solidity is not particularly well-suited to generate this behavior, resulting in a large
465:         // amount of generated bytecode.
466: 
467:         if (msg.sender != address(this)) {
468:             // We perform an external call to ourselves, forwarding the same calldata. In this call, the else clause of
469:             // the preceding if statement will be executed instead.
470: 
471:             // solhint-disable-next-line avoid-low-level-calls
472:             (bool success, ) = address(this).call(msg.data);
473: 
474:             // solhint-disable-next-line no-inline-assembly
475:             assembly {
476:                 // This call should always revert to decode the actual asset deltas from the revert reason
477:                 switch success
478:                     case 0 {
479:                         // Note we are manually writing the memory slot 0. We can safely overwrite whatever is
480:                         // stored there as we take full control of the execution and then immediately return.
481: 
482:                         // We copy the first 4 bytes to check if it matches with the expected signature, otherwise
483:                         // there was another revert reason and we should forward it.
484:                         returndatacopy(0, 0, 0x04)
485:                         let error := and(mload(0), 0xffffffff00000000000000000000000000000000000000000000000000000000)
486: 
487:                         // If the first 4 bytes don't match with the expected signature, we forward the revert reason.
488:                         if eq(eq(error, 0xfa61cc1200000000000000000000000000000000000000000000000000000000), 0) {
489:                             returndatacopy(0, 0, returndatasize())
490:                             revert(0, returndatasize())
491:                         }
492: 
493:                         // The returndata contains the signature, followed by the raw memory representation of an array:
494:                         // length + data. We need to return an ABI-encoded representation of this array.
495:                         // An ABI-encoded array contains an additional field when compared to its raw memory
496:                         // representation: an offset to the location of the length. The offset itself is 32 bytes long,
497:                         // so the smallest value we  can use is 32 for the data to be located immediately after it.
498:                         mstore(0, 32)

522:                 // length (stored at `deltas`) by 32.
