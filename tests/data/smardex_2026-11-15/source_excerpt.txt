 114       * @return firstAmountIn_ the first amount of in-token
 115       *
 116       * @dev if the trade is going in the direction that the price will never reach the priceAverage, or if _amountIn
 117       * is not big enough to reach the priceAverage or if the price is already equal to the priceAverage, then

 120      function computeFirstTradeQtyIn(GetAmountParameters memory _param) internal pure returns (uint256 firstAmountIn_) {
 121          // default value
 122          firstAmountIn_ = _param.amount;
 123  
 124          // if trade is in the good direction
 125          if (_param.fictiveReserveOut * _param.priceAverageIn > _param.fictiveReserveIn * _param.priceAverageOut) {
 126              // pre-compute all operands
 127              uint256 _toSub = _param.fictiveReserveIn * ((FEES_BASE * 2) - (_param.feesPool * 2) - _param.feesLP);
 128              uint256 _toDiv = (FEES_BASE - _param.feesPool) * 2;
 129              uint256 _inSqrt = (((_param.fictiveReserveIn * _param.fictiveReserveOut) * 4) / _param.priceAverageOut) *
 130                  _param.priceAverageIn *
 131                  ((FEES_BASE - _param.feesPool - _param.feesLP) * (FEES_BASE - _param.feesPool)) +
 132                  ((_param.fictiveReserveIn * _param.fictiveReserveIn) * (_param.feesLP * _param.feesLP));
 133  
 134              // reverse sqrt check to only compute sqrt if really needed
 135              uint256 _inSqrtCompare = _toSub + _param.amount * _toDiv;
 136              if (_inSqrt < _inSqrtCompare * _inSqrtCompare) {
 137                  firstAmountIn_ = (Math.sqrt(_inSqrt) - _toSub) / _toDiv;
 138              }
