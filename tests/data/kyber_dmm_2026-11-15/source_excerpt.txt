 285      function verifyBalanceAndUpdateEma(
 286          uint256 amount0In,
 287          uint256 amount1In,
 288          uint256 beforeReserve0,
 289          uint256 beforeReserve1,
 290          uint256 afterReserve0,
 291          uint256 afterReserve1
 292      ) internal virtual returns (uint256 feeInPrecision) {
 293          // volume = beforeReserve0 * amount1In / beforeReserve1 + amount0In (normalized into amount in token 0)
 294          uint256 volume = beforeReserve0.mul(amount1In).div(beforeReserve1).add(amount0In);
 295          uint256 rFactorInPrecision = recordNewUpdatedVolume(block.number, volume);
 296          feeInPrecision = getFinalFee(FeeFomula.getFee(rFactorInPrecision), ampBps);
 297          // verify balance update matches with fomula
 298          uint256 balance0Adjusted = afterReserve0.mul(PRECISION);
 299          balance0Adjusted = balance0Adjusted.sub(amount0In.mul(feeInPrecision));
 300          balance0Adjusted = balance0Adjusted / PRECISION;
 301          uint256 balance1Adjusted = afterReserve1.mul(PRECISION);
 302          balance1Adjusted = balance1Adjusted.sub(amount1In.mul(feeInPrecision));
 303          balance1Adjusted = balance1Adjusted / PRECISION;
 304          require(
 305              balance0Adjusted.mul(balance1Adjusted) >= beforeReserve0.mul(beforeReserve1),
 306              "DMM: K"
 307          );
 308      }

  50      function recordNewUpdatedVolume(uint256 blockNumber, uint256 value)
  51          internal
  52          returns (uint256)
  53      {
  54          // this can not be underflow because block.number always increases
  55          uint256 skipBlock = blockNumber - lastTradeBlock;
  56          if (skipBlock == 0) {
  57              currentBlockVolume = safeUint128(
  58                  uint256(currentBlockVolume).add(value),
  59                  "volume exceeds valid range"
  60              );
  61              return calculateRFactor(uint256(shortEMA), uint256(longEMA));
  62          }
  63          uint128 _currentBlockVolume = currentBlockVolume;
  64          uint256 _shortEMA = newEMA(shortEMA, SHORT_ALPHA, currentBlockVolume);
  65          uint256 _longEMA = newEMA(longEMA, LONG_ALPHA, currentBlockVolume);
  66          // ema = ema * (1-aplha) ^(skipBlock -1)
  67          _shortEMA = _shortEMA.mulInPrecision(
  68              (PRECISION - SHORT_ALPHA).unsafePowInPrecision(skipBlock - 1)
  69          );
  70          _longEMA = _longEMA.mulInPrecision(
  71              (PRECISION - LONG_ALPHA).unsafePowInPrecision(skipBlock - 1)
  72          );
  73          shortEMA = safeUint128(_shortEMA);
  74          longEMA = safeUint128(_longEMA);
  75          currentBlockVolume = safeUint128(value);
  76          lastTradeBlock = safeUint128(blockNumber);
  77  
  78          emit UpdateEMA(_shortEMA, _longEMA, _currentBlockVolume, skipBlock);
  79  
  80          return calculateRFactor(_shortEMA, _longEMA);
  81      }

  83      /// @return rFactor in Precision for this trade
  84      function getRFactor(uint256 blockNumber) internal view returns (uint256) {
  85          // this can not be underflow because block.number always increases
  86          uint256 skipBlock = blockNumber - lastTradeBlock;
  87          if (skipBlock == 0) {
  88              return calculateRFactor(shortEMA, longEMA);
  89          }
  90          uint256 _shortEMA = newEMA(shortEMA, SHORT_ALPHA, currentBlockVolume);
  91          uint256 _longEMA = newEMA(longEMA, LONG_ALPHA, currentBlockVolume);
  92          _shortEMA = _shortEMA.mulInPrecision(
  93              (PRECISION - SHORT_ALPHA).unsafePowInPrecision(skipBlock - 1)
  94          );
  95          _longEMA = _longEMA.mulInPrecision(
  96              (PRECISION - LONG_ALPHA).unsafePowInPrecision(skipBlock - 1)
  97          );
  98          return calculateRFactor(_shortEMA, _longEMA);
